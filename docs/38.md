Create the CI workflow at the repository root:

```zsh
mkdir -p .github/workflows
nvim .github/workflows/ci.yml
```

Add:

```yaml
name: CI

on:
  push:
    branches:
      - main
  pull_request:
  workflow_dispatch:

permissions:
  contents: read

concurrency:
  group: ci-${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

jobs:
  static-checks:
    name: Static checks
    runs-on: ubuntu-24.04
    timeout-minutes: 10

    steps:
      - name: Check out repository
        uses: actions/checkout@v5

      - name: Check shell syntax
        run: |
          find deploy/self-hosted/scripts \
            -type f \
            -name '*.sh' \
            -exec sh -n {} +

      - name: Prepare non-secret production environment
        working-directory: deploy/self-hosted
        run: cp .env.production.example .env.production

      - name: Validate production Compose
        working-directory: deploy/self-hosted
        run: |
          docker compose \
            --env-file .env.production \
            --file docker-compose.yml \
            --file docker-compose.production.yml \
            --file docker-compose.caddy.yml \
            config --quiet

      - name: Ensure production environment remains ignored
        run: |
          git check-ignore \
            deploy/self-hosted/.env.production

      - name: Check repository formatting
        run: git diff --check

  database:
    name: Database migrations and tests
    runs-on: ubuntu-24.04
    timeout-minutes: 20

    steps:
      - name: Check out repository
        uses: actions/checkout@v5

      - name: Install Supabase CLI
        uses: supabase/setup-cli@v3
        with:
          version: 2.115.0

      - name: Confirm CLI version
        run: |
          test "$(supabase --version)" = "2.115.0"

      - name: Start fresh Supabase database
        run: supabase db start

      - name: Run database tests
        run: supabase test db

      - name: Run database advisors
        run: supabase db advisors

      - name: Show migration status
        run: supabase migration list --local

      - name: Stop Supabase
        if: always()
        run: supabase stop --no-backup
```

This workflow has no production credentials and cannot deploy anything.

## Test equivalent commands locally

From the repository root:

```zsh
find deploy/self-hosted/scripts \
  -type f \
  -name '*.sh' \
  -exec sh -n {} +

supabase db reset
supabase test db
supabase db advisors
supabase migration list --local
```

Validate the template configuration:

```zsh
cd deploy/self-hosted

cp .env.production.example /tmp/personal-supabase-production.env

docker compose \
  --env-file /tmp/personal-supabase-production.env \
  --file docker-compose.yml \
  --file docker-compose.production.yml \
  --file docker-compose.caddy.yml \
  config --quiet

cd ../..
```

## Review and commit

```zsh
git diff --check
git diff -- .github/workflows/ci.yml

git add .github/workflows/ci.yml
git commit -m "Add Supabase continuous integration"
git push
```

Then open the repository’s GitHub Actions page and verify both jobs:

```text
Static checks
Database migrations and tests
```

The workflow uses the current official `supabase/setup-cli@v3` action while pinning your CLI to `2.115.0`. [Official setup action](https://github.com/supabase/setup-cli).

If CI passes, we’ll add a deployment workflow template that is manual-only, environment-protected, and intentionally incapable of running until VPS secrets are configured.
