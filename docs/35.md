Now we turn the successful manual procedure into a production backup script. It creates portable database dumps plus a Storage archive and briefly closes the public gateway to keep database metadata and file bytes consistent.

## 1. Create the script

```zsh
nvim scripts/backup.sh
```

Add:

```sh
#!/usr/bin/env sh

set -eu

cd "$(dirname "$0")/.."

ENV_FILE=".env.production"
BACKUP_ROOT="${BACKUP_ROOT:-../../backups}"
BACKUP_ID="$(date -u +%Y%m%dT%H%M%SZ)"
BACKUP_DIR="$BACKUP_ROOT/$BACKUP_ID"

compose() {
  docker compose \
    --env-file "$ENV_FILE" \
    --file docker-compose.yml \
    --file docker-compose.production.yml \
    --file docker-compose.caddy.yml \
    "$@"
}

if [ ! -f "$ENV_FILE" ]; then
  echo "Missing $ENV_FILE" >&2
  exit 1
fi

command -v supabase >/dev/null 2>&1 || {
  echo "Supabase CLI is required" >&2
  exit 1
}

command -v docker >/dev/null 2>&1 || {
  echo "Docker is required" >&2
  exit 1
}

mkdir -p "$BACKUP_DIR"
BACKUP_DIR="$(cd "$BACKUP_DIR" && pwd -P)"

POOLER_TENANT_ID="$(
  sed -n 's/^POOLER_TENANT_ID=//p' "$ENV_FILE"
)"

POSTGRES_PASSWORD="$(
  sed -n 's/^POSTGRES_PASSWORD=//p' "$ENV_FILE"
)"

if [ -z "$POOLER_TENANT_ID" ] || [ -z "$POSTGRES_PASSWORD" ]; then
  echo "Database configuration is incomplete" >&2
  exit 1
fi

DB_URL="postgresql://postgres.${POOLER_TENANT_ID}:${POSTGRES_PASSWORD}@127.0.0.1:5432/postgres?sslmode=disable"

STORAGE_CONTAINER="$(compose ps --quiet storage)"

if [ -z "$STORAGE_CONTAINER" ]; then
  echo "Storage container is not running" >&2
  exit 1
fi

STORAGE_VOLUME="$(
  docker inspect "$STORAGE_CONTAINER" \
    --format '{{range .Mounts}}{{if eq .Destination "/var/lib/storage"}}{{.Name}}{{end}}{{end}}'
)"

if [ -z "$STORAGE_VOLUME" ]; then
  echo "Could not determine the Storage volume" >&2
  exit 1
fi

# Pull before maintenance begins so an image download cannot extend downtime.
docker image inspect alpine:3.22 >/dev/null 2>&1 ||
  docker pull alpine:3.22

SERVICES_STOPPED=true

restore_services() {
  if [ "$SERVICES_STOPPED" = true ]; then
    compose start storage functions api-gw caddy >/dev/null 2>&1 || true
  fi
}

trap restore_services EXIT INT TERM

echo "Entering backup maintenance mode..."

compose stop caddy api-gw storage functions

echo "Creating database dumps..."

supabase db dump \
  --db-url "$DB_URL" \
  --file "$BACKUP_DIR/roles.sql" \
  --role-only

supabase db dump \
  --db-url "$DB_URL" \
  --file "$BACKUP_DIR/schema.sql"

supabase db dump \
  --db-url "$DB_URL" \
  --file "$BACKUP_DIR/data.sql" \
  --use-copy \
  --data-only

supabase db dump \
  --db-url "$DB_URL" \
  --file "$BACKUP_DIR/migration-schema.sql" \
  --schema supabase_migrations

supabase db dump \
  --db-url "$DB_URL" \
  --file "$BACKUP_DIR/migration-data.sql" \
  --schema supabase_migrations \
  --use-copy \
  --data-only

echo "Archiving Storage..."

docker run --rm \
  --volume "$STORAGE_VOLUME:/source:ro" \
  --volume "$BACKUP_DIR:/backup" \
  alpine:3.22 \
  tar -czf /backup/storage.tar.gz -C /source .

echo "Recording deployment information..."

compose config --images > "$BACKUP_DIR/images.txt"
compose ps > "$BACKUP_DIR/services.txt"

echo "Generating checksums..."

(
  cd "$BACKUP_DIR"

  shasum -a 256 \
    roles.sql \
    schema.sql \
    data.sql \
    migration-schema.sql \
    migration-data.sql \
    storage.tar.gz \
    images.txt \
    > SHA256SUMS
)

echo "Leaving maintenance mode..."

compose start storage functions api-gw caddy
SERVICES_STOPPED=false

unset POSTGRES_PASSWORD DB_URL

echo "Backup completed:"
echo "$BACKUP_DIR"
```

## 2. Make it executable and validate syntax

```zsh
chmod +x scripts/backup.sh
sh -n scripts/backup.sh
```

## 3. Do not run it against the local stack yet

The script intentionally uses:

```text
.env.production
docker-compose.production.yml
docker-compose.caddy.yml
```

It is intended for the future production deployment. Your current local environment already has a successfully rehearsed backup.

## 4. Document the future invocation

On the VPS:

```zsh
scripts/backup.sh
```

A different backup destination can be supplied without editing the script:

```zsh
BACKUP_ROOT=/var/backups/personal-supabase \
  scripts/backup.sh
```

Eventually the backup root should be copied to another machine or object-storage provider. A backup stored only on the same VPS does not protect against VPS or disk loss.

## 5. Commit

```zsh
git add scripts/backup.sh
git commit -m "Add production backup script"
```
